# Python packages needed to run the pipeline.
#
# Versions are unpinned except XGBoost, whose trees can change between
# releases. The manuscript's models were trained with XGBoost 3.1.2.
# Even with the same version, re-training on a different operating system can
# give slightly different trees (row/column subsampling), so expect test-set
# accuracy to move by a few tenths of a percentage point; see the README.
# To pin everything to the environment that produced a set of results:
#     pip freeze > requirements-lock.txt
# and keep that file alongside the results.

# --- core ---
pyyaml             # project_paths.yaml / pinned_runs.yaml
numpy
pandas
scipy
pyarrow            # parquet read/write

# --- gridded and geospatial data ---
xarray
rioxarray          # the .rio accessor used when regridding
netCDF4            # writing the hourly predictor cubes
rasterio
geopandas
shapely
pyproj
contextily         # basemap tiles for the station and study-extent maps
matplotlib-scalebar

# --- interpolation and modelling ---
pykrige            # ordinary and universal kriging
scikit-learn
xgboost==3.1.2
betacal            # beta calibration of the predicted probabilities
shap               # per-prediction feature attribution

# --- plotting ---
matplotlib
seaborn

# --- misc ---
pytz               # station timezone conversion
tqdm               # progress bars during interpolation
